POST /payment_intents

Initializes a new payment intent with customer amount, currency, and payment method options. 

import requests

url = "https://api.paygateway.com/v2/payment_intents"
headers = {
    "Authorization": "Bearer YOUR_ACCESS_TOKEN",
    "Content-Type": "application/json"
}
payload = {
    "amount": 2500,
    "currency": "usd",
    "customer_id": "cus_88921a"
}

response = requests.post(url, headers=headers, json=payload)
print(response.json())
